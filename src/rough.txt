Skin Profile Page:

Concerns - Multi Choice
    1.Acne, 2.Dark Spots, 3.Dryness, 4.Aging, 5.Oily Skin, 6.Dullness

Skin Type - Multi Choice
    1.Oily, 2.Normal, 3.Dry, 4.Sensitive, 5.Combination, 6.Not Sure

Skincare Routine - Single Choice
    1.Every day, 2.Rarely, 3.Sometimes, 4.Never

Allergies - Yes or No 
    If yes, text box

Goal - Multi Choice
    1.Clearer Skin, 2.Smoother Texture, 3.Brighter Glow, 4.Anti-Aging, 5.Fever Pimples

Menstrual Cycle
    Q1-next cycle start
        1.Next few days, 2.2+ weeks later, 3.Next week, 4.Not Sure
    Q2-how skin behave
        1.Become more sensitive, 2.Gets oily, 3.Not Sure, 4.Sligh acne, 5.No major change
    Q3-reminders
        1.Yes, 2.No

Other Symptoms - Yes or No
    If Yes, text box















import { useState } from 'react';
import { FiEdit, FiSave, FiX } from 'react-icons/fi';

const SkinProfilePage = () => {
  const [isEditing, setIsEditing] = useState(false);
  const [skinData, setSkinData] = useState({
    concerns: ['Acne', 'Dark Spots'],
    skinType: ['Combination', 'Sensitive'],
    skincareRoutine: 'Every day',
    allergies: {
      hasAllergies: false,
      details: 'Fragrance, Essential Oils'
    },
    goals: ['Clearer Skin', 'Brighter Glow'],
    menstrualCycle: {
      nextCycle: 'Next week',
      skinBehavior: 'Gets oily',
      reminders: true
    },
    otherSymptoms: {
      hasSymptoms: true,
      details: 'Occasional redness and irritation'
    }
  });

  const [tempData, setTempData] = useState(JSON.parse(JSON.stringify(skinData)));

  // Available options
  const concernOptions = ['Acne', 'Dark Spots', 'Dryness', 'Aging', 'Oily Skin', 'Dullness'];
  const skinTypeOptions = ['Oily', 'Normal', 'Dry', 'Sensitive', 'Combination', 'Not Sure'];
  const routineOptions = ['Every day', 'Rarely', 'Sometimes', 'Never'];
  const cycleOptions = ['Next few days', '2+ weeks later', 'Next week', 'Not Sure'];
  const skinBehaviorOptions = ['Become more sensitive', 'Gets oily', 'Not Sure', 'Slight acne', 'No major change'];
  const goalOptions = ['Clearer Skin', 'Smoother Texture', 'Brighter Glow', 'Anti-Aging', 'Fever Pimples'];

  const handleEdit = () => {
    setTempData(JSON.parse(JSON.stringify(skinData)));
    setIsEditing(true);
  };

  const handleSave = () => {
    setSkinData(JSON.parse(JSON.stringify(tempData)));
    setIsEditing(false);
  };

  const handleCancel = () => {
    setTempData(JSON.parse(JSON.stringify(skinData)));
    setIsEditing(false);
  };

  const handleMultiSelect = (field, value) => {
    setTempData(prev => {
      const currentArray = prev[field] || [];
      const newArray = currentArray.includes(value)
        ? currentArray.filter(item => item !== value)
        : [...currentArray, value];
      return {
        ...prev,
        [field]: newArray
      };
    });
  };

  const handleSingleSelect = (field, value) => {
    setTempData(prev => ({
      ...prev,
      [field]: value
    }));
  };

  const handleNestedSelect = (parent, field, value) => {
    setTempData(prev => ({
      ...prev,
      [parent]: {
        ...prev[parent],
        [field]: value
      }
    }));
  };

  const handleAllergiesChange = (hasAllergies) => {
    setTempData(prev => ({
      ...prev,
      allergies: {
        hasAllergies,
        details: hasAllergies ? prev.allergies.details : ''
      }
    }));
  };

  const handleSymptomsChange = (hasSymptoms) => {
    setTempData(prev => ({
      ...prev,
      otherSymptoms: {
        hasSymptoms,
        details: hasSymptoms ? prev.otherSymptoms.details : ''
      }
    }));
  };

  const renderMultiSelect = (label, field, options, currentValues) => (
    <div className="space-y-2">
      <label className="block text-sm font-semibold text-gray-800">
        {label}
      </label>
      {isEditing ? (
        <div className="grid grid-cols-2 gap-1 max-h-32 overflow-y-auto">
          {options.map(option => (
            <label key={option} className="flex items-center space-x-2 p-1 hover:bg-purple-50 rounded">
              <input
                type="checkbox"
                checked={currentValues.includes(option)}
                onChange={() => handleMultiSelect(field, option)}
                className="w-3 h-3 text-[#994A97] border-gray-300 rounded focus:ring-[#994A97]"
              />
              <span className="text-xs text-gray-700">{option}</span>
            </label>
          ))}
        </div>
      ) : (
        <div className="flex flex-wrap gap-1">
          {currentValues.length > 0 ? (
            currentValues.map(value => (
              <span
                key={value}
                className="inline-block bg-[#994A97] text-white px-3 py-1 rounded-full text-sm font-medium"
              >
                {value}
              </span>
            ))
          ) : (
            <span className="text-gray-500 italic text-xs">Not specified</span>
          )}
        </div>
      )}
    </div>
  );

  const renderSingleSelect = (label, field, options, currentValue) => (
    <div className="space-y-2">
      <label className="block text-sm font-semibold text-gray-800">
        {label}
      </label>
      {isEditing ? (
        <div className="space-y-1">
          {options.map(option => (
            <label key={option} className="flex items-center space-x-2 p-1 hover:bg-purple-50 rounded">
              <input
                type="radio"
                checked={currentValue === option}
                onChange={() => handleSingleSelect(field, option)}
                className="w-3 h-3 text-[#994A97] border-gray-300 focus:ring-[#994A97]"
              />
              <span className="text-xs text-gray-700">{option}</span>
            </label>
          ))}
        </div>
      ) : (
        <span className="inline-block bg-[#994A97] text-white px-3 py-1 rounded-full text-sm font-medium">
          {currentValue}
        </span>
      )}
    </div>
  );

  const renderYesNoSection = (label, field, hasValue, details, onChange) => (
    <div className="space-y-2">
      <label className="block text-sm font-semibold text-gray-800">
        {label}
      </label>
      {isEditing ? (
        <div className="space-y-2">
          <div className="flex gap-3">
            <label className="flex items-center space-x-1">
              <input
                type="radio"
                checked={hasValue}
                onChange={() => onChange(true)}
                className="w-3 h-3 text-[#994A97] border-gray-300 focus:ring-[#994A97]"
              />
              <span className="text-xs text-gray-700">Yes</span>
            </label>
            <label className="flex items-center space-x-1">
              <input
                type="radio"
                checked={!hasValue}
                onChange={() => onChange(false)}
                className="w-3 h-3 text-[#994A97] border-gray-300 focus:ring-[#994A97]"
              />
              <span className="text-xs text-gray-700">No</span>
            </label>
          </div>
          {hasValue && (
            <textarea
              value={details}
              onChange={(e) => setTempData(prev => ({
                ...prev,
                [field]: {
                  ...prev[field],
                  details: e.target.value
                }
              }))}
              rows="2"
              className="w-full px-2 py-1 text-xs border border-gray-300 rounded focus:outline-none focus:ring-1 focus:ring-[#994A97]"
              placeholder={`Specify ${label.toLowerCase()}...`}
            />
          )}
        </div>
      ) : (
        <div>
            {hasValue ? 
            ( <div></div> ) :
            (
          <span className={`inline-block px-3 py-1 rounded-full text-sm font-medium ${hasValue ? 'bg-[#994A97] text-white' : 'bg-gray-200 text-gray-700'}`}>
            No
          </span>
          ) }
          {hasValue && details && (
            <p className="text-sm text-gray-600 mt-1 bg-gray-50 px-2 py-1 rounded">
              {details}
            </p>
          )}
        </div>
      )}
    </div>
  );

  return (
    <div className="min-h-screen bg-[#f7bff5] py-4 px-3 sm:px-4">
      <div className="max-w-7xl mx-auto">
        
        <div className="flex justify-between items-center mb-6">
          <h1 className="text-2xl font-bold text-[#994A97]">Skin Profile</h1>
          {!isEditing ? (
            <button
              onClick={handleEdit}
              className="flex items-center gap-2 bg-[#994A97] text-white px-3 py-1.5 rounded-lg hover:bg-[#883885] transition-colors text-sm"
            >
              <FiEdit className="w-3 h-3" />
              Edit Profile
            </button>
          ) : (
            <div className="flex gap-2">
              <button
                onClick={handleCancel}
                className="flex items-center gap-2 bg-gray-500 text-white px-3 py-1.5 rounded-lg hover:bg-gray-600 transition-colors text-sm"
              >
                <FiX className="w-3 h-3" />
                Cancel
              </button>
              <button
                onClick={handleSave}
                className="flex items-center gap-2 bg-[#CA88B1] text-white px-3 py-1.5 rounded-lg hover:bg-[#BA78A1] transition-colors text-sm"
              >
                <FiSave className="w-3 h-3" />
                Save
              </button>
            </div>
          )}
        </div>

        {/* Main Content */}
        <div className="grid grid-cols-1 lg:grid-cols-10 gap-4">
          {/* Left Side - 70% */}
          <div className="lg:col-span-7 space-y-4">
            {/* 2x2 Grid for Concerns, Skin Type, Goals, Routine */}
            <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
              {/* Concerns */}
              <div className="bg-white rounded-xl shadow-sm p-4 border-l-4 border-[#994A97]">
                {renderMultiSelect('Concerns', 'concerns', concernOptions, tempData.concerns)}
              </div>

              {/* Skin Type */}
              <div className="bg-white rounded-xl shadow-sm p-4 border-l-4 border-[#CA88B1]">
                {renderMultiSelect('Skin Type', 'skinType', skinTypeOptions, tempData.skinType)}
              </div>

              {/* Goals */}
              <div className="bg-white rounded-xl shadow-sm p-4 border-l-4 border-[#994A97]">
                {renderMultiSelect('Goals', 'goals', goalOptions, tempData.goals)}
              </div>

              {/* Skincare Routine */}
              <div className="bg-white rounded-xl shadow-sm p-4 border-l-4 border-[#CA88B1]">
                {renderSingleSelect('Skincare Routine', 'skincareRoutine', routineOptions, tempData.skincareRoutine)}
              </div>

            {/* Allergies */}
            <div className="bg-white rounded-xl shadow-sm p-4 border-l-4 border-[#994A97]">
              {renderYesNoSection(
                  'Allergies',
                  'allergies',
                  tempData.allergies.hasAllergies,
                  tempData.allergies.details,
                  handleAllergiesChange
                )}
            </div>

            <div className="bg-white rounded-xl shadow-sm p-4 border-l-4 border-[#994A97]">
              {renderYesNoSection(
                  'Other Symptoms',
                  'otherSymptoms',
                  tempData.otherSymptoms.hasSymptoms,
                  tempData.otherSymptoms.details,
                  handleSymptomsChange
                )}
            </div>
                </div>

            {/* Menstrual Cycle */}
            <div className="bg-white rounded-xl shadow-sm p-4 border-l-4 border-[#CA88B1]">
              <h3 className="text-base font-semibold text-[#994A97] mb-3">Menstrual Cycle</h3>
              <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                {/* Next Cycle */}
                <div className="space-y-2">
                  <label className="block text-sm font-semibold text-gray-800">
                    Next Cycle Start
                  </label>
                  {isEditing ? (
                    <div className="space-y-1">
                      {cycleOptions.map(option => (
                        <label key={option} className="flex items-center space-x-2 p-1 hover:bg-purple-50 rounded">
                          <input
                            type="radio"
                            checked={tempData.menstrualCycle.nextCycle === option}
                            onChange={() => handleNestedSelect('menstrualCycle', 'nextCycle', option)}
                            className="w-3 h-3 text-[#994A97] border-gray-300 focus:ring-[#994A97]"
                          />
                          <span className="text-xs text-gray-700">{option}</span>
                        </label>
                      ))}
                    </div>
                  ) : (
                    <span className="inline-block bg-[#CA88B1] text-white px-3 py-1 rounded-full text-sm font-medium">
                      {tempData.menstrualCycle.nextCycle}
                    </span>
                  )}
                </div>

                {/* Skin Behavior */}
                <div className="space-y-2">
                  <label className="block text-sm font-semibold text-gray-800">
                    Skin Behavior
                  </label>
                  {isEditing ? (
                    <div className="space-y-1 max-h-28 ">
                      {skinBehaviorOptions.map(option => (
                        <label key={option} className="flex items-center space-x-2 p-1 hover:bg-purple-50 rounded">
                          <input
                            type="radio"
                            checked={tempData.menstrualCycle.skinBehavior === option}
                            onChange={() => handleNestedSelect('menstrualCycle', 'skinBehavior', option)}
                            className="w-3 h-3 text-[#994A97] border-gray-300 focus:ring-[#994A97]"
                          />
                          <span className="text-xs text-gray-700">{option}</span>
                        </label>
                      ))}
                    </div>
                  ) : (
                    <span className="inline-block bg-[#994A97] text-white px-3 py-1 rounded-full text-sm font-medium">
                      {tempData.menstrualCycle.skinBehavior}
                    </span>
                  )}
                </div>

                {/* Reminders */}
                <div className="space-y-2 md:col-span-2">
                  <label className="block text-sm font-semibold text-gray-800">
                    Cycle Reminders
                  </label>
                  {isEditing ? (
                    <div className="flex gap-3">
                      <label className="flex items-center space-x-1">
                        <input
                          type="radio"
                          checked={tempData.menstrualCycle.reminders}
                          onChange={() => handleNestedSelect('menstrualCycle', 'reminders', true)}
                          className="w-3 h-3 text-[#994A97] border-gray-300 focus:ring-[#994A97]"
                        />
                        <span className="text-xs text-gray-700">Yes</span>
                      </label>
                      <label className="flex items-center space-x-1">
                        <input
                          type="radio"
                          checked={!tempData.menstrualCycle.reminders}
                          onChange={() => handleNestedSelect('menstrualCycle', 'reminders', false)}
                          className="w-3 h-3 text-[#994A97] border-gray-300 focus:ring-[#994A97]"
                        />
                        <span className="text-xs text-gray-700">No</span>
                      </label>
                    </div>
                  ) : (
                    <span className={`inline-block px-3 py-1 rounded-full text-sm font-medium ${tempData.menstrualCycle.reminders ? 'bg-[#994A97] text-white' : 'bg-gray-200 text-gray-700'}`}>
                      {tempData.menstrualCycle.reminders ? 'Yes' : 'No'}
                    </span>
                  )}
                </div>
              </div>
            </div>


            
          </div>

          
          <div className="lg:col-span-3">
            <div className="bg-linear-to-b h-80 m-6 from-[#994A97] to-[#CA88B1] rounded-xl shadow-sm p-4 sticky top-4 text-white">
           
            </div>
          </div>
        </div>
      </div>
    </div>
  );
};

export default SkinProfilePage;
